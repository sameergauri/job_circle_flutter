Future<void> save() async {
    var clientPayout = "";
    if (isFlat) {
      clientPayout = "Flat";
    } else if (isSlab) {
      clientPayout = "Slab";
    } else if (isCtc) {
      clientPayout = "CTC Based";
    }

    var partnerPay = "";
    if (isPFlat) {
      partnerPay = "Flat";
    } else if (isPSlab) {
      partnerPay = "Slab";
    } else if (isPCtc) {
      partnerPay = "CTC Based";
    }

    var payClause = "";
    if (is30) {
      payClause = "30 Days";
    } else if (is60) {
      payClause = "60 Days";
    } else if (is90) {
      payClause = "90 Days";
    } else if (is120) {
      payClause = "45 Days";
    }

    Commercial commercial = Commercial(
        id: widget.commercial!.id,
        companyId: widget.commercial!.companyId!.toInt(),
        process: widget.commercial!.process.toString(),
        natureOfWork: widget.commercial!.natureOfWork.toString(),
        roleName: widget.commercial!.roleName.toString(),
        payoutType: clientPayout,
        isPartner: isPartner == true ? 1 : 0,
        // partnerPayoutType: partnerPayout.text != null ? partnerPayout.text : null,
        paymentClause: payClause,
        specialPaymentAmount: double.tryParse(spcPA.text),
        specialPaymentClause: spcPC.text != null ? spcPC.text : null,
        commercialActive: _isActive ? 1 : 0,
        endDate: widget.commercial!.endDate != null
            ? DateTime.parse(widget.commercial!.endDate.toString())
            : null,
        ctcPercent: double.tryParse(precent.text),
        partnerPercent: isPartner == true
            ? 0.6 * (double.tryParse(precent.text) ?? 0)
            : isPartner == false && partnerPrecent.text.isNotEmpty
                ? double.tryParse(partnerPrecent.text) ?? 0
                : isPartner == false && partnerPrecent.text.isEmpty
                    ? 0
                    : 0, // Handle any other case if needed

        isConfirm: widget.commercial!.isConfirm,
        amount: double.tryParse(clientA.text),
        partnerAmount: isPartner == true
            ? 0.6 * (double.tryParse(clientA.text) ?? 0)
            : isPartner == false && partnerA.text.isNotEmpty
                ? double.tryParse(partnerA.text) ?? 0
                : isPartner == false && partnerA.text.isEmpty
                    ? 0
                    : 0, // Handle any other case if needed

        startDate: singleSelect != null
            ? singleSelect
            : (widget.commercial!.startDate != null &&
                    widget.commercial!.startDate is String)
                ? DateTime.parse(widget.commercial!.startDate!)
                : DateTime.now(),
        partnerPayoutType: partnerPay);

    List<Slab> combinedSlab = [];

    for (int i = 0; i < client.length; i++) {
      List<dynamic>? slabIdsDynamic = widget.commercial?.slabId;
      List<int>? slabIds = slabIdsDynamic?.map<int>((dynamic value) {
        if (value is int) {
          return value;
        } else if (value is String) {
          return int.tryParse(value) ?? 0;
        }
        return 0; // Default value if neither int nor String
      }).toList();
      int slabId = slabIds != null && i < slabIds.length ? slabIds[i] : 0;

      Slab slab = Slab(
        id: slabId,
        minCount: int.tryParse(client[i].minCountController.text),
        maxCount: client[i].maxCountController.text,
        slabAmount: int.tryParse(client[i].slabAmountController.text),
        isPercent: 0,
        isFresher: 0,
        isCtcSlab: 0,
      );

      // if (i < partner.length) {
      //   slab.partnerMinCount =
      //       int.tryParse(partner[i].partnerMinCountController.text);
      //   slab.partnerMaxCount = partner[i].partnerMaxCountController.text;
      //   slab.partnerSlabAmount =
      //       int.tryParse(partner[i].partnerSlabAmountController.text);
      // }

      if (i < partner.length) {
        slab.partnerMinCount = isPartner == true
            ? (0.6 * (double.tryParse(client[i].minCountController.text) ?? 0))
                .toInt()
            : isPartner == false &&
                    partner[i].partnerMinCountController.text.isNotEmpty
                ? int.tryParse(partner[i].partnerMinCountController.text) ?? 0
                : 0;
        slab.partnerMaxCount = isPartner == true
            ? (0.6 * (double.tryParse(client[i].maxCountController.text) ?? 0))
                .toString()
            : isPartner == false &&
                    partner[i].partnerMaxCountController.text.isNotEmpty
                ? partner[i].partnerMaxCountController.text
                : '0';
        slab.partnerSlabAmount = isPartner == true
            ? (0.6 *
                    (double.tryParse(client[i].slabAmountController.text) ?? 0))
                .toInt()
            : isPartner == false &&
                    partner[i].partnerSlabAmountController.text.isNotEmpty
                ? int.tryParse(partner[i].partnerSlabAmountController.text) ?? 0
                : 0;
      }

      combinedSlab.add(slab);
    }

    CommmercialAddModel commercialModel = CommmercialAddModel(
      slab: combinedSlab,
      commercial: commercial,
    );

    // Convert the companyModel to a JSON representation
    Map<String, dynamic> requestBody = commercialModel.toJson();

    try {
      final response = await http.post(
        Uri.parse('http://${GlobalConstants.API_Host}/commercial/v1/save'),
        headers: {
          'Content-Type': 'application/json',
        },
        body: jsonEncode(requestBody),
      );

      if (response.statusCode == 200) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Company saved successfully')),
        );
        Navigator.pop(context);
      } else {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Failed to save company')),
        );
      }
    } catch (e) {
      print('Error saving data: $e');
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('An error occurred')),
      );
    }
    ref.refresh(commercialProvider);
  }
